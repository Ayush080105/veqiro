### Task A — Parallelize AI Email Processing

**File:** `apps/ai/agents/vega/routes.py`  
**Function:** `process_inbox()`

**Root cause:** The current loop processes emails one at a time:
```python
for email in emails[:request.max_emails]:
    raw = await _llm.complete(...)  # one call per email, sequential
```
With 20 emails this is 20 serial LLM calls — slow.

**Fix:** Extract to async helper, use `asyncio.gather()`:
```python
async def _analyze_email(email: dict, system: str) -> ProcessedEmail:
    raw = await _llm.complete(
        provider=_agent.default_provider, model=_agent.default_model,
        system=system,
        messages=[{"role": "user", "content": (
            "Analyze this email. Return ONLY a JSON object (no markdown fences) with keys: "
            "priority (urgent/high/medium/low), summary (1-2 sentences), "
            "suggested_action (string), label (one of: {label_list}), "
            "hidden_tasks (list of strings), "
            "suggested_reply (string — 1-3 sentence reply suggestion if suggested_action is 'reply', else null), "
            "meeting_request (object with keys date, time, topic if meeting requested, else null)\n\n"
            f"From: {email.get('from', '')}\n"
            f"Subject: {email.get('subject', '')}\n"
            f"Body: {email.get('body', email.get('snippet', ''))[:500]}"
        )}],
    )
    # parse raw → return ProcessedEmail (same logic as current loop body)
    ...

# In process_inbox():
system = await _agent.build_system_prompt(...)
results = await asyncio.gather(*[
    _analyze_email(email, system)
    for email in emails[:request.max_emails]
])
processed = list(results)
```

Also: accept `custom_labels` param (see Task C) and substitute into prompt.

### Task B — Server-side Inbox Cache (10-min TTL)

**Root cause:** Every inbox load re-processes all emails via the AI service, even if nothing changed.

**Step B1 — New Prisma model** in `apps/server/prisma/schema.prisma`:
```prisma
model VegaInboxCache {
  id             String   @id @default(cuid())
  organizationId String   @unique
  snapshot       Json
  cachedAt       DateTime @default(now())

  @@map("vega_inbox_cache")
}
```
Run `npx prisma migrate dev --name add_vega_inbox_cache` after adding.

**Step B2 — Update `getInbox()`** in `apps/server/src/modules/agents/vega/vega.workspace.service.ts`:

```ts
export const getInbox = async (
  userId: string,
  organizationId: string,
  input: z.infer<typeof getInboxSchema>,
  force = false  // new param
): Promise<WorkspaceInboxResponse> => {

  // Check cache first (unless force=true)
  if (!force) {
    const cached = await prisma.vegaInboxCache.findUnique({
      where: { organizationId }
    })
    if (cached) {
      const ageMs = Date.now() - cached.cachedAt.getTime()
      if (ageMs < 10 * 60 * 1000) {  // 10 minutes
        return cached.snapshot as WorkspaceInboxResponse
      }
    }
  }

  // ... existing fetch + process logic ...

  // Upsert cache
  await prisma.vegaInboxCache.upsert({
    where: { organizationId },
    update: { snapshot: result as object, cachedAt: new Date() },
    create: { organizationId, snapshot: result as object },
  })

  return result
}
```

```ts
const DEFAULT_LABELS = ["Investors", "Sales Leads", "Newsletters", "Team", "Legal", "Finance", "Other"]

export const getLabels = async (organizationId: string) => {
  // Seed defaults if org has no labels yet
  const count = await prisma.vegaLabel.count({ where: { organizationId } })
  if (count === 0) {
    await prisma.vegaLabel.createMany({
      data: DEFAULT_LABELS.map((name) => ({ name, organizationId })),
      skipDuplicates: true,
    })
  }
  return prisma.vegaLabel.findMany({
    where: { organizationId },
    orderBy: { createdAt: "asc" },
  })
}

export const createLabel = async (organizationId: string, input: { name: string; color?: string }) => {
  return prisma.vegaLabel.create({
    data: { name: input.name, color: input.color ?? "#999999", organizationId },
  })
}

export const deleteLabel = async (organizationId: string, labelId: string) => {
  const existing = await prisma.vegaLabel.findFirst({ where: { id: labelId, organizationId } })
  if (!existing) throw new BadRequestError("Label not found")
  await prisma.vegaLabel.delete({ where: { id: labelId } })
}

export const updateLabel = async (
  organizationId: string,
  labelId: string,
  input: { autoReply?: boolean; color?: string }
) => {
  const existing = await prisma.vegaLabel.findFirst({ where: { id: labelId, organizationId } })
  if (!existing) throw new BadRequestError("Label not found")
  return prisma.vegaLabel.update({
    where: { id: labelId },
    data: { autoReply: input.autoReply, color: input.color },
  })
}
```

```ts
export async function draftVegaReply(
```

#### F1 — Implement email.job.ts

Full file implementation for `apps/server/src/modules/agents/vega/email.job.ts`:

```ts
import { aiService } from "../../../common/utils/aiService.js"
import { getGoogleAccessToken, GoogleNotConnectedError } from "../../../common/utils/googleAuth.js"
import { labelMessage, createGmailDraft } from "../../../common/utils/googleApis.js"
import { prisma } from "../../../config/prisma.js"
import { getLabels } from "./vega.workspace.service.js"

export async function runEmailPipeline(userId: string, organizationId: string): Promise<void> {
  // 1. Get Google token
  let token: string
  try {
    token = await getGoogleAccessToken(userId)
  } catch (err) {
    if (err instanceof GoogleNotConnectedError) return  // skip orgs without Google
    throw err
  }

  // 2. Get org's labels (with autoReply flags)
  const labels = await getLabels(organizationId)
  const labelNames = labels.map(l => l.name)
  const autoReplyLabels = new Set(labels.filter(l => l.autoReply).map(l => l.name))

  // 3. Process inbox via AI
  const { data } = await aiService.post("/ai/vega/process-inbox", {
    user_id: userId,
    organization_id: organizationId,
    max_emails: 20,
    auto_label: true,
    draft_replies: false,
    custom_labels: labelNames,
    metadata: { google_access_token: token },
  })

  const processed = data.processed ?? []

  // 4. Execute label_messages node_actions (apply labels in Gmail)
  const labelAction = (data.node_actions ?? []).find(
    (a: { node_action: string }) => a.node_action === "label_messages"
  )
  if (labelAction?.messages) {
    await Promise.allSettled(
      labelAction.messages.map(({ email_id, label }: { email_id: string; label: string }) =>
        labelMessage({ accessToken: token, messageId: email_id, labelName: `Vega/${label}` })
          .catch((err: unknown) => console.error(`[email-job] Label failed for ${email_id}:`, err))
      )
    )
  }

  // 5. Auto-draft replies for matching labels
  const autoReplyEmails = processed.filter(
    (e: { label_applied?: string }) => e.label_applied && autoReplyLabels.has(e.label_applied)
  )

  await Promise.allSettled(
    autoReplyEmails.map(async (e: { email_id: string; label_applied: string }) => {
      try {
        const { data: draftData } = await aiService.post("/ai/vega/draft-reply", {
          user_id: userId,
          organization_id: organizationId,
          email_id: e.email_id,
          reply_instructions: "Draft a professional contextual reply",
          tone: "professional",
          save_as_draft: true,
          metadata: { google_access_token: token },
        })

        const draftAction = (draftData.node_actions ?? []).find(
          (a: { node_action: string }) => a.node_action === "create_gmail_draft"
        )
        if (draftAction) {
          await createGmailDraft({
            accessToken: token,
            to: draftAction.to,
            subject: draftAction.subject,
            body: draftAction.body,
            replyToMessageId: draftAction.reply_to_message_id ?? null,
            replyToThreadId: draftAction.reply_to_thread_id ?? null,
          })
        }
      } catch (err) {
        console.error(`[email-job] Auto-draft failed for email ${e.email_id}:`, err)
      }
    })
  )

  // 6. Update inbox cache
  await prisma.vegaInboxCache.upsert({
    where: { organizationId },
    update: { snapshot: { emails: processed } as object, cachedAt: new Date() },
    create: { organizationId, snapshot: { emails: processed } as object },
  })

  console.log(`[email-job] Processed ${processed.length} emails for org ${organizationId}`)
}
```
